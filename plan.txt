function buildGantt() {
  const root = document.getElementById('gantt-root');
  root.innerHTML = '';

  // Year / quarter / month headers
  const yearRow = document.createElement('div');
  yearRow.style.cssText = 'display:grid;grid-template-columns:220px 90px repeat(36,1fr);background:var(--surface3);border-bottom:1px solid var(--border);position:sticky;top:82px;z-index:51;';

  // Legislation + Risk cols
  const blankCell = (text='', extra='') => `<div style="border-right:1px solid var(--border);padding:6px 10px;font-family:'DM Mono',monospace;font-size:9px;letter-spacing:0.1em;text-transform:uppercase;color:var(--text-muted);${extra}">${text}</div>`;
  yearRow.innerHTML = blankCell('Legislation / Activity') + blankCell('Risk / LoD', 'text-align:center;border-right:2px solid var(--border-light);');

  for (let y=0;y<3;y++) {
    const ycell = document.createElement('div');
    ycell.style.cssText = `grid-column:span 12;border-right:2px solid var(--border-light);padding:6px 8px;font-family:'DM Mono',monospace;font-size:10px;font-weight:500;color:var(--amber);letter-spacing:0.15em;text-align:center;background:var(--surface3);`;
    ycell.textContent = YEARS[y];
    yearRow.appendChild(ycell);
  }
  root.appendChild(yearRow);

  // Month row
  const monthRow = document.createElement('div');
  monthRow.style.cssText = 'display:grid;grid-template-columns:220px 90px repeat(36,1fr);border-bottom:2px solid var(--border);background:var(--surface2);position:sticky;top:120px;z-index:50;';
  monthRow.innerHTML = blankCell('','') + blankCell('','border-right:2px solid var(--border-light);');
  for (let m=0;m<36;m++) {
    const mi = m % 12;
    const qcls = QTR_COLOR[mi];
    const isYearEnd = (m % 12 === 11);
    monthRow.innerHTML += `<div style="border-right:${isYearEnd ? '2px solid var(--border-light)' : '1px solid var(--border)'};padding:5px 2px;font-family:'DM Mono',monospace;font-size:8px;letter-spacing:0.05em;color:${qcls==='q1'?'#60a5fa':qcls==='q2'?'#34d399':qcls==='q3'?'#f59e0b':'#f87171'};text-align:center;">${MONTH_SHORT[mi]}</div>`;
  }
  root.appendChild(monthRow);

  // Legislation groups
  LEGISLATIONS.forEach((leg, li) => {
    const group = document.createElement('div');
    group.style.cssText = `border-bottom:2px solid var(--border-light);`;

    // Legislation header row
    const hrow = document.createElement('div');
    hrow.style.cssText = `display:grid;grid-template-columns:220px 90px repeat(36,1fr);background:var(--surface2);border-bottom:1px solid var(--border);cursor:pointer;`;
    hrow.onclick = () => {
      group.classList.toggle('collapsed');
    };

    const riskMeta = RISK_LEVELS[leg.risk];
    hrow.innerHTML = `
      <div class="sticky-col" style="padding:10px 14px;display:flex;align-items:center;gap:8px;border-right:1px solid var(--border);background:var(--surface2);">
        <span style="font-size:12px;color:var(--text-muted);">▾</span>
        <div>
          <div style="font-family:'Fraunces',serif;font-size:13px;font-weight:600;color:var(--text-primary)">${leg.tag}</div>
          <div style="font-size:9px;color:var(--text-muted);margin-top:2px">${leg.full.substring(0,55)}…</div>
        </div>
      </div>
      <div class="sticky-col2" style="padding:10px 8px;display:flex;align-items:center;justify-content:center;border-right:2px solid var(--border-light);background:var(--surface2);">
        <span class="risk-badge ${riskMeta.cls}">${riskMeta.label}</span>
      </div>
    `;

    // Empty gantt cells for header row (subtle shading)
    for (let m=0;m<36;m++) {
      const isYE = (m%12===11);
      const cell = document.createElement('div');
      cell.style.cssText = `border-right:${isYE?'2px solid var(--border-light)':'1px solid var(--border)'};background:rgba(30,37,53,0.3);`;
      hrow.appendChild(cell);
    }
    group.appendChild(hrow);

    // Activity rows
    const body = document.createElement('div');
    body.className = 'collapsible-body';

    leg.activities.forEach(act => {
      const arow = document.createElement('div');
      arow.style.cssText = `display:grid;grid-template-columns:220px 90px repeat(36,1fr);border-bottom:1px solid rgba(42,51,71,0.5);min-height:32px;`;

      const lodMeta = LOD[act.lod];
      const typeMeta = TYPES[act.type];

      arow.innerHTML = `
        <div class="sticky-col" style="padding:5px 14px 5px 24px;display:flex;align-items:center;gap:6px;border-right:1px solid var(--border);background:var(--surface);">
          <div style="width:3px;border-radius:1px;align-self:stretch;background:var(--l${act.lod});min-height:14px;flex-shrink:0;"></div>
          <span style="font-size:10px;color:var(--text-secondary);flex:1;">${act.name}</span>
          <span class="act-type-badge" style="background:${typeMeta.color}22;border:1px solid ${typeMeta.color}66;color:${typeMeta.color};font-family:'DM Mono',monospace;font-size:7px;padding:1px 4px;border-radius:2px;white-space:nowrap;">${typeMeta.label.split(' ')[0]}</span>
        </div>
        <div class="sticky-col2" style="padding:5px 8px;display:flex;align-items:center;justify-content:center;border-right:2px solid var(--border-light);background:var(--surface);">
          <span class="lod-tag ${lodMeta.cls}" style="font-family:'DM Mono',monospace;font-size:8px;padding:2px 5px;border-radius:2px;">${lodMeta.label}</span>
        </div>
      `;

      // Build bar lookup: which months are covered
      const barAtMonth = {};
      act.bars.forEach(b => {
        for (let m=b.start; m<b.start+b.dur; m++) {
          if (!barAtMonth[m]) barAtMonth[m] = b;
        }
      });

      for (let m=0;m<36;m++) {
        const isYE = (m%12===11);
        const cell = document.createElement('div');
        cell.style.cssText = `border-right:${isYE?'2px solid var(--border-light)':'1px solid rgba(42,51,71,0.35)'};padding:3px 1px;display:flex;align-items:center;position:relative;`;

        const b = barAtMonth[m];
        if (b) {
          const tm = TYPES[b.type];
          const isStart = (m === b.start);
          const isEnd   = (m === b.start + b.dur - 1);
          const barDiv = document.createElement('div');
          let br = isStart && isEnd ? '3px' : isStart ? '3px 0 0 3px' : isEnd ? '0 3px 3px 0' : '0';
          barDiv.style.cssText = `height:14px;border-radius:${br};width:100%;background:${tm.color}26;border:1px solid ${tm.color}66;color:${tm.color};display:flex;align-items:center;justify-content:center;font-family:'DM Mono',monospace;font-size:6px;font-weight:500;overflow:hidden;`;
          if (isStart && b.label) barDiv.textContent = b.label;
          barDiv.title = `${tm.label} — ${act.name}`;
          barDiv.addEventListener('mouseenter', e => showTooltip(e, tm.label, act.name, YEARS[Math.floor(m/12)], lodMeta.label));
          barDiv.addEventListener('mousemove', moveTooltip);
          barDiv.addEventListener('mouseleave', hideTooltip);
          cell.appendChild(barDiv);
        }
        arow.appendChild(cell);
      }

      body.appendChild(arow);
    });

    group.appendChild(body);
    root.appendChild(group);
  });
}

function buildHeatmap() {
  const root = document.getElementById('heatmap-root');
  root.innerHTML = '';

  const legs = LEGISLATIONS.map(l=>l.tag);
  const lods = ['1','2','3'];

  // Count activities per leg × lod
  const grid = {};
  LEGISLATIONS.forEach(leg => {
    grid[leg.tag] = {total:0};
    lods.forEach(l => grid[leg.tag][l] = 0);
    leg.activities.forEach(a => {
      grid[leg.tag][a.lod]++;
      grid[leg.tag].total++;
    });
  });

  const max = Math.max(...Object.values(grid).map(v=>v.total));

  const container = document.createElement('div');
  container.style.cssText = 'display:grid;grid-template-columns:180px repeat('+legs.length+',1fr);gap:3px;';

  // Header
  container.innerHTML = `<div style="font-family:'DM Mono',monospace;font-size:9px;color:var(--text-muted);padding:6px;display:flex;align-items:flex-end;">Legislation</div>`;
  legs.forEach(l => {
    container.innerHTML += `<div style="font-family:'DM Mono',monospace;font-size:8px;color:var(--text-secondary);text-align:center;padding:4px 2px;writing-mode:vertical-lr;transform:rotate(180deg);height:90px;display:flex;align-items:center;justify-content:center;">${l}</div>`;
  });

  // Rows per LoD
  const lodLabels = {
    '1': 'Line 1 — Business',
    '2': 'Line 2 — Compliance',
    '3': 'Line 3 — Audit',
    'total': 'TOTAL'
  };

  [...lods, 'total'].forEach(lod => {
    container.innerHTML += `<div style="font-size:10px;color:${lod==='1'?'var(--l1)':lod==='2'?'var(--l2)':lod==='3'?'var(--l3)':'var(--text-secondary)'};display:flex;align-items:center;padding:4px 6px;font-family:'DM Mono',monospace;font-weight:${lod==='total'?'600':'400'};border-top:${lod==='total'?'1px solid var(--border)':'none'}">${lodLabels[lod]}</div>`;
    legs.forEach(l => {
      const val = grid[l][lod] || 0;
      const intensity = val===0?0:Math.ceil(val/max*6);
      container.innerHTML += `<div class="hm-val hm-${intensity}" style="border-top:${lod==='total'?'1px solid var(--border)':'none'}">${val||'—'}</div>`;
    });
  });

  root.appendChild(container);

  // Build distribution chart
  buildDist();
}

function buildDist() {
  const root = document.getElementById('dist-root');
  root.innerHTML = '';

  const typeOrder = ['ct','att','tr','aud','mon','crev','reg'];
  const counts = {};
  typeOrder.forEach(t => counts[t]={});
  LEGISLATIONS.forEach(leg => {
    typeOrder.forEach(t => counts[t][leg.tag]=0);
    leg.activities.forEach(a => {
      counts[a.type][leg.tag] = (counts[a.type][leg.tag]||0) + a.bars.length;
    });
  });

  const legs = LEGISLATIONS.map(l=>l.tag);
  const container = document.createElement('div');
  container.style.cssText = 'overflow-x:auto;';

  const table = document.createElement('div');
  table.style.cssText = `display:grid;grid-template-columns:160px repeat(${legs.length},1fr);gap:2px;min-width:800px;`;

  table.innerHTML = `<div style="font-family:'DM Mono',monospace;font-size:9px;color:var(--text-muted);padding:6px;display:flex;align-items:center;">Activity Type</div>`;
  legs.forEach(l => {
    table.innerHTML += `<div style="font-family:'DM Mono',monospace;font-size:8px;color:var(--text-secondary);text-align:center;padding:4px;border-bottom:1px solid var(--border);">${l}</div>`;
  });

  typeOrder.forEach(t => {
    const tm = TYPES[t];
    table.innerHTML += `<div style="display:flex;align-items:center;gap:6px;padding:6px;font-size:10px;color:${tm.color};"><span style="width:8px;height:8px;background:${tm.color};border-radius:2px;display:inline-block;flex-shrink:0;"></span>${tm.label}</div>`;
    legs.forEach(l => {
      const val = counts[t][l]||0;
      table.innerHTML += `<div style="display:flex;align-items:center;justify-content:center;background:${val?tm.color+'1a':'var(--surface2)'};border-radius:3px;font-family:'DM Mono',monospace;font-size:10px;color:${val?tm.color:'var(--text-muted)'};min-height:28px;">${val||'—'}</div>`;
    });
  });

  container.appendChild(table);
  root.appendChild(container);
}

const tt = document.getElementById('tooltip');
function showTooltip(e, type, name, year, lod) {
  document.getElementById('tt-title').textContent = type;
  document.getElementById('tt-body').innerHTML = `<div style="color:var(--text-secondary);margin-bottom:4px;">${name}</div><div style="font-family:'DM Mono',monospace;font-size:9px;color:var(--text-muted);">${year} · ${lod}</div>`;
  tt.style.display='block';
  moveTooltip(e);
}
function moveTooltip(e) {
  tt.style.left = (e.clientX+14)+'px';
  tt.style.top  = (e.clientY-10)+'px';
}
function hideTooltip() { tt.style.display='none'; }

function switchTab(id) {
  document.querySelectorAll('.tab').forEach((t,i) => {
    const ids = ['gantt','heatmap','summary','governance'];
    t.classList.toggle('active', ids[i]===id);
  });
  document.querySelectorAll('.view-section').forEach(s => s.classList.remove('active'));
  document.getElementById('tab-'+id).classList.add('active');
}
